cmake_minimum_required(VERSION 3.13)

set(PICO_BOARD pico2_w CACHE STRING "Target board")
if(NOT PICO_BOARD STREQUAL "pico2_w")
    message(FATAL_ERROR "This wiring example requires PICO_BOARD=pico2_w")
endif()
if(NOT PICO_SDK_PATH AND DEFINED ENV{PICO_SDK_PATH})
    set(PICO_SDK_PATH "$ENV{PICO_SDK_PATH}")
endif()
if(NOT EXISTS "${PICO_SDK_PATH}/external/pico_sdk_import.cmake")
    message(FATAL_ERROR "Set PICO_SDK_PATH to a Pico SDK 2.3.x checkout")
endif()
include("${PICO_SDK_PATH}/external/pico_sdk_import.cmake")

project(display_layouts C CXX ASM)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
pico_sdk_init()
if(NOT PICO_SDK_VERSION_STRING MATCHES "^2\\.3\\.")
    message(FATAL_ERROR "Use Pico SDK 2.3.x for this example")
endif()

include(assets.cmake)
add_executable(display_layouts main.c demo.c layouts.c
    ../display-hello/epd.c "${LAYOUT_ASSET_SOURCE}")
target_include_directories(display_layouts PRIVATE . ../display-hello)
target_compile_options(display_layouts PRIVATE -Wall -Wextra -Werror)
target_link_libraries(display_layouts pico_stdlib hardware_spi)
pico_enable_stdio_usb(display_layouts 1)
pico_enable_stdio_uart(display_layouts 0)
pico_add_extra_outputs(display_layouts)
